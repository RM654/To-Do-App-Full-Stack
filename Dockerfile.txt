FROM node:18-alpine as build

WORKDIR /app
COPY package.json yarn.lock ./
RUN yarn install
COPY . .
RUN yarn build   # builds into `dist` or `build` depending on your setup

# Use a lightweight static server image
FROM nginx:stable-alpine
COPY --from=build /app/build /usr/share/nginx/html
# Optionally copy a custom nginx.conf to route /api to backend, or rely on docker-compose network + reverse proxy

EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]
